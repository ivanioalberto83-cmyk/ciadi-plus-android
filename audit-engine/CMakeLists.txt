cmake_minimum_required(VERSION 3.16)
project(ciadi_audit_engine LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(ciadi-audit-engine src/main.cpp)
if (MSVC)
  target_compile_options(ciadi-audit-engine PRIVATE /W4)
else()
  target_compile_options(ciadi-audit-engine PRIVATE -Wall -Wextra -Wpedantic)
endif()
